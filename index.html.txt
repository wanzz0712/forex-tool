            <button id="tabForex" onclick="setMode('forex')" class="flex-1 py-2 rounded-lg text-sm font-bold tab-active transition">貨幣對 (FX)</button>

            <div>
                <label class="block text-gray-400 text-xs uppercase mb-2">帳戶本金 (USD)</label>
                <input type="number" id="balance" value="10000" class="w-full py-4 rounded-xl text-2xl font-mono focus:ring-2 focus:ring-blue-500 outline-none">
            </div>

            <div>
                <label class="block text-gray-400 text-xs uppercase mb-2">單筆風險: <span id="riskValue" class="text-blue-400">1%</span></label>
                <input type="range" id="riskPct" min="0.5" max="5" step="0.5" value="1" class="w-full h-2 bg-gray-700 rounded-lg appearance-none cursor-pointer accent-blue-500">
            </div>

            <div>
                <label id="slLabel" class="block text-gray-400 text-xs uppercase mb-2">止損距離 (Pips/點)</label>
                <input type="number" id="slInput" value="30" class="w-full py-4 rounded-xl text-2xl font-mono focus:ring-2 focus:ring-blue-500 outline-none">
                <p id="slHint" class="text-[10px] text-gray-500 mt-2 text-center">提示：1 手黃金波動 1 美金 = 100 點</p>
            </div>

            <div id="resultCard" class="bg-green-500/10 border border-green-500/50 rounded-2xl p-6 text-center mt-8">
                <p class="text-gray-400 text-xs mb-1 uppercase">建議下單規模</p>
                <h2 id="finalLots" class="text-6xl font-black text-green-400">0.33</h2>
                <p class="text-sm mt-2 text-gray-300 font-bold">Standard Lots</p>
                
                <div class="grid grid-cols-2 gap-2 mt-6 border-t border-gray-700/50 pt-4">
                    <div class="text-left">
                        <p class="text-[10px] text-gray-500 uppercase">預計虧損</p>
                        <p id="loseAmount" class="font-bold text-white text-lg">$100</p>
                    </div>
                    <div class="text-right">
                        <p class="text-[10px] text-gray-500 uppercase">風險評級</p>
                        <p id="riskLevel" class="font-bold text-green-400 text-lg">安全</p>
                    </div>
                </div>
            </div>

    <script>
        let mode = 'forex'; // 'forex' or 'gold'
        const balIn = document.getElementById('balance');
        const riskIn = document.getElementById('riskPct');
        const slIn = document.getElementById('slInput');
        const riskValDisp = document.getElementById('riskValue');
        
        function setMode(m) {
            mode = m;
            document.getElementById('tabForex').className = m === 'forex' ? 'flex-1 py-2 rounded-lg text-sm font-bold tab-active' : 'flex-1 py-2 rounded-lg text-sm font-bold tab-inactive';
            document.getElementById('tabGold').className = m === 'gold' ? 'flex-1 py-2 rounded-lg text-sm font-bold tab-active' : 'flex-1 py-2 rounded-lg text-sm font-bold tab-inactive';
            
            // 調整提示字
            document.getElementById('slLabel').innerText = m === 'forex' ? '止損距離 (Pips/點)' : '止損距離 (點/Points)';
            document.getElementById('slHint').innerText = m === 'forex' ? '提示：EUR/USD 等主流幣 10 點 = 1 Pip' : '提示：黃金 100 點 = 1 美金波動';
            
            calculate();
        }

        function calculate() {
            const bal = parseFloat(balIn.value) || 0;
            const rPct = parseFloat(riskIn.value);
            const sl = parseFloat(slIn.value) || 1;
            
            riskValDisp.innerText = rPct + '%';
            const riskMoney = bal * (rPct / 100);

            let lots = 0;
            if (mode === 'forex') {
                // 一般貨幣對: 1 標準手, 1 Pip (10點) = $10 USD
                lots = riskMoney / (sl * 10);
            } else {
                // 黃金 (XAU/USD): 1 標準手, 1 點 (0.01美金) = $1 USD
                lots = riskMoney / sl;
            }

            // 更新 UI
            document.getElementById('finalLots').innerText = lots.toFixed(2);
            document.getElementById('loseAmount').innerText = '$' + riskMoney.toLocaleString();

            const resCard = document.getElementById('resultCard');
            const resLots = document.getElementById('finalLots');
            const resLv = document.getElementById('riskLevel');

            if (rPct <= 2) {
                resLv.innerText = "安全";
                resLv.className = "font-bold text-green-400 text-lg";
                resCard.className = "bg-green-500/10 border border-green-500/50 rounded-2xl p-6 text-center mt-8";
                resLots.className = "text-6xl font-black text-green-400";
            } else if (rPct <= 3.5) {
                resLv.innerText = "謹慎";
                resLv.className = "font-bold text-yellow-400 text-lg";
                resCard.className = "bg-yellow-500/10 border border-yellow-500/50 rounded-2xl p-6 text-center mt-8";
                resLots.className = "text-6xl font-black text-yellow-400";
            } else {
                resLv.innerText = "危險";
                resLv.className = "font-bold text-red-400 text-lg";
                resCard.className = "bg-red-500/10 border border-red-500/50 rounded-2xl p-6 text-center mt-8";
                resLots.className = "text-6xl font-black text-red-400";
            }
        }

        [balIn, riskIn, slIn].forEach(el => el.addEventListener('input', calculate));
        calculate();
    </script>
